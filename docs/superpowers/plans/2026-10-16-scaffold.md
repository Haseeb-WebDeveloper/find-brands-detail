# Brand Enrich — Step 1: Scaffold — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A deployable Next.js app where a reviewer logs in with the shared password, pastes brand names, and watches each brand move through the seven research steps live (steps are placeholders in this plan; later plans fill them in).

**Architecture:** Next.js 16 App Router on Vercel; Supabase Postgres accessed server-side through Drizzle; Inngest runs one durable job per brand; the browser receives live row changes through Supabase Realtime using a short-lived JWT the server mints after password login. Row-level security lets that JWT read and nothing else; all writes go through server code.

**Tech Stack:** TypeScript 5, Next.js ^16.3.5, React ^19.3.0, Tailwind ^4.1.0, Zod ^4.6.5, Drizzle ORM ^0.45.2 + drizzle-kit ^0.31.10, postgres ^3.4.9, @supabase/supabase-js ^2.116.0, Inngest ^4.20.0, jose ^6, Vitest ^3.2.0, pnpm 11.

**Spec:** `docs/superpowers/specs/2026-10-16-brand-enrich-design.md` (§4 Architecture, §6 Data model and flow, §8 Screens, §9 Errors). Read it before starting.

## Global Constraints

- Package manager: pnpm (`packageManager: "pnpm@11.5.0"`); dependency versions as listed in Tech Stack.
- Users: one shared password (`APP_PASSWORD`); no individual accounts (spec §3).
- Brand status values exactly: `queued`, `researching`, `needs_review`, `approved`, `exported` (spec §6).
- Research steps, in order, exactly: `website`, `crawl`, `instagram`, `tiktok`, `images`, `campaigns`, `score` (spec §6).
- The public Supabase anon key alone must read nothing; reads require the server-minted JWT with claim `app_role = "reviewer"` (spec §8).
- Writes happen only in server code through `DATABASE_URL` (spec §8).
- Grant table privileges explicitly per table; never rely on `ALTER DEFAULT PRIVILEGES`.
- Secrets are never logged, printed, or committed. `.env*.local` stays git-ignored.
- Every commit message ends with:
  ```

  ```

- [ ] **Step 3: Implement `src/lib/env.ts`**

```ts
/**
 * Named env access. A missing variable must fail with its own name, not as `undefined`
 * surfacing three calls later as a confusing library error.
 */
export function requireEnv(name: string): string {
  const value = process.env[name];
  if (!value || value.trim() === "") {
    throw new Error(`${name} is not set. See .env.example.`);
  }
  return value;
}

export function requireSecret(name: string, minLength = 32): string {
  const value = requireEnv(name);
  if (value.length < minLength) {
    throw new Error(`${name} must be at least ${minLength} characters.`);
  }
  return value;
}
```

```ts
import {
  boolean, index, integer, jsonb, pgEnum, pgTable, primaryKey, text, timestamp, uuid,
} from "drizzle-orm/pg-core";

export const batchStatus = pgEnum("batch_status", ["active", "archived"]);
export const brandStatus = pgEnum("brand_status", ["queued", "researching", "needs_review", "approved", "exported"]);
export const fieldName = pgEnum("field_name", ["website", "instagram", "tiktok", "logo", "cover"]);
export const candidateScore = pgEnum("candidate_score", ["proven", "review", "rejected"]);
export const chosenBy = pgEnum("chosen_by", ["auto", "reviewer"]);
export const campaignStatus = pgEnum("campaign_status", ["proposed", "approved", "removed"]);
export const imageSourceKind = pgEnum("image_source_kind", ["brand", "third_party"]);

export type BrandStatus = (typeof brandStatus.enumValues)[number];

const createdAt = () => timestamp("created_at", { withTimezone: true }).notNull().defaultNow();

export const batches = pgTable("batches", {
  id: uuid("id").primaryKey().defaultRandom(),
  status: batchStatus("status").notNull().default("active"),
  createdAt: createdAt(),
});

export const brands = pgTable(
  "brands",
  {
    id: uuid("id").primaryKey().defaultRandom(),
    batchId: uuid("batch_id").notNull().references(() => batches.id, { onDelete: "cascade" }),
    name: text("name").notNull(),
    websiteHint: text("website_hint"),
    status: brandStatus("status").notNull().default("queued"),
    currentStep: text("current_step"),
    reviewCount: integer("review_count").notNull().default(0),
    error: text("error"),
    createdAt: createdAt(),
    updatedAt: timestamp("updated_at", { withTimezone: true }).notNull().defaultNow(),
  },
  (t) => [index("brands_batch_id_idx").on(t.batchId)],
);

export const candidates = pgTable(
  "candidates",
  {
    id: uuid("id").primaryKey().defaultRandom(),
    brandId: uuid("brand_id").notNull().references(() => brands.id, { onDelete: "cascade" }),
    field: fieldName("field").notNull(),
    value: text("value").notNull(),
    source: text("source").notNull(),
    evidence: jsonb("evidence").$type<Record<string, unknown>>().notNull().default({}),
    score: candidateScore("score").notNull(),
    reason: text("reason").notNull(),
    createdAt: createdAt(),
  },
  (t) => [index("candidates_brand_id_idx").on(t.brandId)],
);

export const selections = pgTable(
  "selections",
  {
    brandId: uuid("brand_id").notNull().references(() => brands.id, { onDelete: "cascade" }),
    field: fieldName("field").notNull(),
    candidateId: uuid("candidate_id").references(() => candidates.id, { onDelete: "set null" }),
    typedValue: text("typed_value"),
    chosenBy: chosenBy("chosen_by").notNull(),
  },
  (t) => [primaryKey({ columns: [t.brandId, t.field] })],
);

export const campaigns = pgTable(
  "campaigns",
  {
    id: uuid("id").primaryKey().defaultRandom(),
    brandId: uuid("brand_id").notNull().references(() => brands.id, { onDelete: "cascade" }),
    position: integer("position").notNull(),
    title: text("title").notNull(),
    description: text("description").notNull(),
    sourceUrls: jsonb("source_urls").$type<string[]>().notNull().default([]),
    status: campaignStatus("status").notNull().default("proposed"),
    createdAt: createdAt(),
  },
  (t) => [index("campaigns_brand_id_idx").on(t.brandId)],
);

export const campaignImages = pgTable(
  "campaign_images",
  {
    id: uuid("id").primaryKey().defaultRandom(),
    campaignId: uuid("campaign_id").notNull().references(() => campaigns.id, { onDelete: "cascade" }),
    url: text("url").notNull(),
    sourceKind: imageSourceKind("source_kind").notNull(),
    width: integer("width"),
    height: integer("height"),
    score: candidateScore("score").notNull(),
    reason: text("reason").notNull(),
    selected: boolean("selected").notNull().default(false),
  },
  (t) => [index("campaign_images_campaign_id_idx").on(t.campaignId)],
);

export const brandText = pgTable("brand_text", {
  brandId: uuid("brand_id").primaryKey().references(() => brands.id, { onDelete: "cascade" }),
  description: text("description"),
  tier: text("tier"),
  aesthetic: text("aesthetic"),
  industry: text("industry"),
  regions: text("regions"),
  updatedAt: timestamp("updated_at", { withTimezone: true }).notNull().defaultNow(),
});
```

```ts
import { createHmac, timingSafeEqual } from "node:crypto";
import { requireSecret } from "@/lib/env";

export const SESSION_COOKIE = "be_session";
export const SESSION_MAX_AGE_SECONDS = 60 * 60 * 24 * 7;

function sign(expiry: number): string {
  return createHmac("sha256", requireSecret("SESSION_SECRET")).update(`v1.${expiry}`).digest("base64url");
}

export function createSessionToken(now = Math.floor(Date.now() / 1000)): string {
  const expiry = now + SESSION_MAX_AGE_SECONDS;
  return `${expiry}.${sign(expiry)}`;
}

export function verifySessionToken(token: string | undefined, now = Math.floor(Date.now() / 1000)): boolean {
  if (!token) return false;
  const parts = token.split(".");
  if (parts.length !== 2) return false;
  const expiry = Number(parts[0]);
  if (!Number.isInteger(expiry) || expiry <= now) return false;
  const expected = Buffer.from(sign(expiry));
  const actual = Buffer.from(parts[1]);
  return expected.length === actual.length && timingSafeEqual(expected, actual);
}
```

```ts
import { NextResponse, type NextRequest } from "next/server";
import { isPublicPath } from "@/lib/auth/public-paths";
import { SESSION_COOKIE, verifySessionToken } from "@/lib/auth/session";

export function proxy(request: NextRequest) {
  const { pathname, search } = request.nextUrl;
  if (isPublicPath(pathname)) return NextResponse.next();
  if (verifySessionToken(request.cookies.get(SESSION_COOKIE)?.value)) return NextResponse.next();

  if (pathname.startsWith("/api/")) {
    return NextResponse.json({ error: "Not signed in" }, { status: 401 });
  }
  const url = request.nextUrl.clone();
  url.pathname = "/login";
  url.search = `?next=${encodeURIComponent(pathname + search)}`;
  return NextResponse.redirect(url);
}
```

```ts
export const REALTIME_TOKEN_TTL_SECONDS = 60 * 60;

/** The only credential that can read data from the browser. RLS checks app_role = 'reviewer'. */
export async function mintRealtimeToken(now = Math.floor(Date.now() / 1000)): Promise<string> {
  return new SignJWT({ role: "authenticated", app_role: "reviewer" })
```

```ts
export const MAX_BRANDS_PER_BATCH = 50;

export type ParsedBrand = { name: string; websiteHint: string | null };

/** A hostname with at least one dot and no spaces, optionally with scheme and path. */
const LOOKS_LIKE_WEBSITE = /^(https?:\/\/)?[a-z0-9-]+(\.[a-z0-9-]+)+(:\d+)?(\/\S*)?$/i;

function toUrl(hint: string): string | null {
  const withScheme = /^https?:\/\//i.test(hint) ? hint : `https://${hint}`;
  try {
    return new URL(withScheme).toString();
  } catch {
    return null;
  }
}

/** Brand names can contain commas ("Dear, Klairs"), so only a trailing part that looks like a website is a hint. */
function splitLine(line: string): ParsedBrand {
  const lastComma = line.lastIndexOf(",");
  if (lastComma > 0) {
    const tail = line.slice(lastComma + 1).trim();
    if (LOOKS_LIKE_WEBSITE.test(tail)) {
      const url = toUrl(tail);
      if (url) return { name: line.slice(0, lastComma).trim(), websiteHint: url };
    }
  }
  return { name: line.trim(), websiteHint: null };
}

export function parseBatchInput(text: string): { brands: ParsedBrand[]; errors: string[] } {
  const brands: ParsedBrand[] = [];
  const errors: string[] = [];
  const seen = new Set<string>();

  text.split(/\r?\n/).forEach((raw, i) => {
    const line = raw.trim();
    if (!line) return;
    const brand = splitLine(line);
    const key = brand.name.toLowerCase();
    if (seen.has(key)) {
      errors.push(`Line ${i + 1}: "${brand.name}" is already in this batch.`);
      return;
    }
    seen.add(key);
    brands.push(brand);
  });

  if (brands.length > MAX_BRANDS_PER_BATCH) {
    return { brands: [], errors: [`At most ${MAX_BRANDS_PER_BATCH} brands per batch.`] };
  }
  return { brands, errors };
}
```

```ts
export const researchBrand = inngest.createFunction(

    triggers: [{ event: EVENTS.brandResearchRequested }],
```

```ts
import "server-only";
import { desc, eq, sql } from "drizzle-orm";
import { db } from "@/lib/db";
import { batches, brands } from "@/lib/db/schema";
import type { ParsedBrand } from "@/lib/batches/parse-input";

export async function createBatch(input: ParsedBrand[]): Promise<{ batchId: string; brandIds: string[] }> {
  return db.transaction(async (tx) => {
    const [batch] = await tx.insert(batches).values({}).returning({ id: batches.id });
    const rows = await tx
      .insert(brands)
      .values(input.map((b) => ({ batchId: batch.id, name: b.name, websiteHint: b.websiteHint })))
      .returning({ id: brands.id });
    return { batchId: batch.id, brandIds: rows.map((r) => r.id) };
  });
}

export async function listBatches() {
  return db
    .select({
      id: batches.id,
      createdAt: batches.createdAt,
      brandCount: sql<number>`count(${brands.id})::int`,
      needsReview: sql<number>`count(*) filter (where ${brands.status} = 'needs_review')::int`,
    })
    .from(batches)
    .leftJoin(brands, eq(brands.batchId, batches.id))
    .groupBy(batches.id)
    .orderBy(desc(batches.createdAt));
}
```

```ts
import type { BrandStatus } from "@/lib/db/schema";
import { RESEARCH_STEPS, stepNumber } from "@/lib/research/steps";

export type LiveBrand = {
  id: string;
  name: string;
  status: BrandStatus;
  currentStep: string | null;
  reviewCount: number;
  error: string | null;
  createdAt: string;
};

type Change = { eventType: "INSERT" | "UPDATE" | "DELETE"; new: Record<string, unknown>; old: Record<string, unknown> };

export function fromRealtimeRow(row: Record<string, unknown>): LiveBrand {
  return {
    id: String(row.id),
    name: String(row.name),
    status: row.status as BrandStatus,
    currentStep: (row.current_step as string | null) ?? null,
    reviewCount: Number(row.review_count ?? 0),
    error: (row.error as string | null) ?? null,
    createdAt: String(row.created_at),
  };
}

const byCreated = (x: LiveBrand, y: LiveBrand) => x.createdAt.localeCompare(y.createdAt);

export function applyBrandChange(rows: LiveBrand[], change: Change): LiveBrand[] {
  if (change.eventType === "DELETE") return rows.filter((r) => r.id !== String(change.old.id));
  const row = fromRealtimeRow(change.new);
  const rest = rows.filter((r) => r.id !== row.id);
  return [...rest, row].sort(byCreated);
}

export function progressText(brand: LiveBrand): string {
  if (brand.error) return brand.error;
  switch (brand.status) {
    case "queued":
      return "Waiting to start";
    case "researching": {
      const step = RESEARCH_STEPS.find((s) => s.id === brand.currentStep);
      return step ? `${step.label} (step ${stepNumber(step.id)} of ${RESEARCH_STEPS.length})` : "Starting";
    }
    case "needs_review":
      return brand.reviewCount > 0 ? `Ready for review · ${brand.reviewCount} need you` : "Ready for review";
    case "approved":
      return "Approved";
    case "exported":
      return "Exported";
  }
}
```

```tsx
import { useEffect, useState } from "react";
import { getBrowserSupabase } from "@/lib/realtime/browser-client";
import { applyBrandChange, progressText, type LiveBrand } from "@/lib/brands/live";

export function BatchLive({ batchId, initial }: { batchId: string; initial: LiveBrand[] }) {
  const [rows, setRows] = useState(initial);
  const [live, setLive] = useState<"connecting" | "live" | "offline">("connecting");

  useEffect(() => {
    const supabase = getBrowserSupabase();
    const channel = supabase
      .channel(`batch:${batchId}`)
      .on(
        "postgres_changes",
        { event: "*", schema: "public", table: "brands", filter: `batch_id=eq.${batchId}` },
        (payload) =>
          setRows((current) =>
            applyBrandChange(current, {
              eventType: payload.eventType,
              new: payload.new as Record<string, unknown>,
              old: payload.old as Record<string, unknown>,
            }),
          ),
      )
      .subscribe((status) => setLive(status === "SUBSCRIBED" ? "live" : status === "CLOSED" || status === "CHANNEL_ERROR" ? "offline" : "connecting"));

    return () => {
      void supabase.removeChannel(channel);
    };
  }, [batchId]);

  return (
    <div className="flex flex-col gap-2">
      <p className="text-xs text-stone-500">
        {live === "live" ? "● Live" : live === "offline" ? "○ Live updates paused — reload the page" : "○ Connecting…"}
```
